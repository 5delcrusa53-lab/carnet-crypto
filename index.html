async function rpc(urls,method,params){ let err; for(const u of [...new Set([].concat(urls).filter(Boolean))]){ const controller=new AbortController(); const timer=setTimeout(()=>controller.abort(),12000); try{ const r=await fetch(u,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({jsonrpc:'2.0',id:1,method,params}),signal:controller.signal}); if(!r.ok)throw new Error('HTTP '+r.status); const d=await r.json();if(d.error)throw new Error(d.error.message||'RPC'); return d.result; }catch(e){err=e} finally{clearTimeout(timer)} } throw err||new Error('RPC'); } const solRpcs=()=>[S.settings.solRpc,'https://solana-rpc.publicnode.com','https://api.mainnet-beta.solana.com','https://rpc.ankr.com/solana'].filter(Boolean); function hexAmount(h,dec){ const b=BigInt(!h||h==='0x'?'0x0':h),s=10n**BigInt(dec); return Number(b/s)+Number(b%s)/Number(s); } async function holdSolana(addr){ const urls=solRpcs(),m=new Map(); let complete=true; try{ const bal=await rpc(urls,'getBalance',[addr]); m.set(CHAINS.solana.native.wrapped,(bal.value||0)/1e9); }catch(e){ complete=false; console.warn('Lecture Solana impossible au niveau principal',e); } for(const prog of TOKEN_PROGS){ try{ const r=await rpc(urls,'getTokenAccountsByOwner',[addr,{programId:prog},{encoding:'jsonParsed'}]); for(const acc of r.value||[]){ const info=acc.account?.data?.parsed?.info;if(!info)continue; const q=+info.tokenAmount.uiAmountString||0; m.set(info.mint,(m.get(info.mint)||0)+q); } }catch(e){ complete=false; console.warn(`Lecture tokens Solana impossible pour ${prog}`,e); } } return {list:[...m].map(([address,qty])=>({address,qty})),complete}; } async function holdEvm(p){ const ch=CHAINS[p.chain],out=[]; const wei=await rpc(ch.rpc,'eth_getBalance',[p.address,'latest']); out.push({address:ch.native.wrapped,qty:hexAmount(wei,18)}); const found=new Set([lc(ch.native.wrapped)]); let complete=false; if(S.settings.moralisKey&&ch.moralis){ try{ const r=await fetch(`https://deep-index.moralis.io/api/v2.2/wallets/${p.address}/tokens?chain=${ch.moralis}&exclude_spam=true`, {headers:{'X-API-Key':S.settings.moralisKey,accept:'application/json'}}); if(!r.ok)throw new Error('Moralis '+r.status); const d=await r.json(); for(const t of d.result||[]){ if(t.native_token||t.possible_spam||!t.token_address)continue; out.push({address:t.token_address,qty:+t.balance_formatted||0,symbol:t.symbol,name:t.name});found.add(lc(t.token_address)); } complete=true; }catch(e){p.syncErr='Clé Moralis refusée ou quota atteint'} } const addrData='0x70a08231'+p.address.slice(2).toLowerCase().padStart(64,'0'); for(const a of assetsOf(p.id)){ if(!a.address||found.has(lc(a.address))||!/^0x[0-9a-fA-F]{40}$/.test(a.address))continue; try{ if(a.decimals==null)a.decimals=parseInt(await rpc(ch.rpc,'eth_call',[{to:a.address,data:'0x313ce567'},'latest']),16)||18; const raw=await rpc(ch.rpc,'eth_call',[{to:a.address,data:addrData},'latest']); out.push({address:a.address,qty:hexAmount(raw,a.decimals)});found.add(lc(a.address)); }catch(e){} } return {list:out,complete}; } function flagMove(p,a,delta){ if(!p.lastSync||histOn(p))return; const ch=CHAINS[a.chain]; if(Math.abs(delta)<=Math.max(1e-9,Math.abs(a.chainQty||0)*1e-6))return; if(a.isNative&&Math.abs(delta)<(ch.native?.min||0))return; const ex=S.pending.find(x=>x.assetId===a.id); if(ex){ex.delta+=delta;ex.ts=Date.now();if(Math.abs(ex.delta)<1e-12)S.pending=S.pending.filter(x=>x!==ex)} else S.pending.push({id:uid(),assetId:a.id,delta,price:curPrice(a),ts:Date.now()}); } function applyHoldings(p,{list,complete}){ const ch=CHAINS[p.chain],seen=new Set(); for(const h of list){ const k=lc(h.address);seen.add(k); let a=S.assets.find(x=>x.portfolioId===p.id&&lc(x.address)===k); if(!a){ if(!(h.qty>0))continue; const isNative=k===lc(ch.native.wrapped); a={id:uid(),portfolioId:p.id,chain:p.chain,address:h.address,isNative,manualPrice:null,auto:true, symbol:isNative?ch.native.symbol:(h.symbol||'?'),name:isNative?ch.native.name:(h.name||''),chainQty:0}; S.assets.push(a); } if(a.chainQty!=null)flagMove(p,a,h.qty-a.chainQty); if(h.symbol&&(!a.symbol||a.symbol==='?')){a.symbol=h.symbol;a.name=h.name||a.name} a.chainQty=h.qty; } if(complete)for(const a of assetsOf(p.id)){ if(a.address&&!seen.has(lc(a.address))&&a.chainQty){flagMove(p,a,-a.chainQty);a.chainQty=0} } p.lastSync=Date.now(); } async function syncAll(){ let err=0; for(const p of S.portfolios.filter(p=>p.sync&&p.address&&p.chain!=='other')){ try{ p.syncErr=null; const result=p.chain==='solana'?await holdSolana(p.address):await holdEvm(p); applyHoldings(p,result); if(!result.complete){ p.syncErr='Solde principal lu, certains tokens sont temporairement indisponibles'; } }catch(e){ p.syncErr='Lecture du wallet impossible : '+(e?.message||'réseau indisponible'); err++; } } return err; }
